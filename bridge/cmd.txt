{
  echo "===== RUN 2 : le marqueur a-t-il survécu ? ====="
  echo "hostname : $(hostname)"
  echo "uptime   : $(uptime -p) (boot: $(uptime -s))"
  echo ""
  if [ -f /tmp/preuve-persistence.txt ]; then

} > bridge/out.txt 2>&1
