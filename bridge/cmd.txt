# ============================================================
# bridge/cmd.txt — VM PERSISTANTE
#   Phase 1 : restauration de bridge/state/snapshot.tar.gz → ~/persist
#   Phase 2 : ZONE COMMANDES UTILISATEUR
#   Phase 3 : snapshot de ~/persist → commit → push (out.txt + état)
# ============================================================

set +e
WS="$HOME/persist"
SNAP="bridge/state/snapshot.tar.gz"

# ---------- PHASE 1 : RESTAURATION DE L'ÉTAT ----------
mkdir -p "$WS"
if [ -f "$SNAP" ]; then
  tar -xzf "$SNAP" -C "$WS" 2>/dev/null
  RESTORED="OUI — $(find "$WS" -type f | wc -l) fichiers, $(du -sh "$WS" | cut -f1)"
else
  RESTORED="NON — première session (aucun snapshot présent)"
fi

# ---------- PHASE 2 : ZONE COMMANDES UTILISATEUR ----------
cat > /tmp/user-zone.sh <<'USERZONE'
cd "$HOME/persist" || exit 1
{
  echo "===== PONT CODESPACE — TEST D'ACCÈS COMPLET ====="
  echo "# host Actions: $(hostname)"
  echo "# date: $(date -u +%FT%TZ)"
  echo "# état restauré : $RESTORED"
  echo ""

  # ---- 0) le secret est-il injecté dans ce run ? ----
  if [ -z "$CODESPACE_TOKEN" ]; then
    echo "❌ ERREUR : CODESPACE_TOKEN vide — le secret n'est pas défini ou pas dans le workflow"
    exit 0
  fi
  echo "✅ secret CODESPACE_TOKEN présent (${#CODESPACE_TOKEN} caractères — valeur masquée)"

  export GH_TOKEN="$CODESPACE_TOKEN"

  echo ""
  echo "--- 1) gh cli ---"
  gh --version | head -1

  echo ""
  echo "--- 2) liste des codespaces du repo ---"
  gh codespace list --repo JeanHug/VM-arena \
    --json name,state,machine \
    --template '{{range .}}{{.name}} | {{.state}} | {{.machine.displayName}}{{"\n"}}{{end}}' 2>&1

  CS=$(gh codespace list --repo JeanHug/VM-arena --json name --jq '.[0].name' 2>/dev/null)
  if [ -z "$CS" ]; then
    echo "❌ aucun codespace trouvé sur JeanHug/VM-arena"
    exit 0
  fi
  echo "→ codespace sélectionné : $CS"

  STATE=$(gh codespace list --repo JeanHug/VM-arena --json name,state --jq '.[0].state' 2>/dev/null)
  echo "→ état actuel : $STATE"

  if [ "$STATE" != "Running" ]; then
    echo ""
    echo "--- 3) démarrage du codespace (30-90 s)… ---"
    if gh codespace start -c "$CS" 2>&1; then
      echo "✅ codespace démarré"
    else
      echo "❌ échec du démarrage"
      exit 0
    fi
  fi

  echo ""
  echo "--- 4) SPECS DU CODESPACE via SSH ---"
  gh codespace ssh -c "$CS" -- 'echo "hostname : $(hostname)"; uname -a; echo "CPU : $(nproc) cœurs"; free -h | head -2; df -h / | tail -1' 2>&1

  echo ""
  echo "--- 5) test d'écriture (marqueur disque persistant) ---"
  gh codespace ssh -c "$CS" -- 'echo "VM-ARENA est passée ici : $(date -u)" > ~/BONJOUR_VM_ARENA.txt && cat ~/BONJOUR_VM_ARENA.txt && ls -la ~/BONJOUR_VM_ARENA.txt' 2>&1

  echo ""
  echo "--- 6) arrêt du codespace (économie du quota, disque conservé) ---"
  gh codespace stop -c "$CS" 2>&1 && echo "✅ arrêté — le disque et le marqueur persistent"
  echo ""
  echo "===== FIN DU TEST ====="
}
exit 0
USERZONE

RESTORED="$RESTORED" timeout 1500 bash /tmp/user-zone.sh > "bridge/out.txt" 2>&1
echo "zone utilisateur terminée (code $?)" >> bridge/out.txt

# ---------- PHASE 3 : SNAPSHOT + PUBLICATION ----------
mkdir -p bridge/state
tar -czf "$SNAP" -C "$WS" . 2>/dev/null
echo "snapshot: $(ls -la "$SNAP" 2>/dev/null || echo 'ECHEC CREATION')" >> bridge/out.txt
git config user.name "vm-bridge"
git config user.email "vm-bridge@users.noreply.github.com"
git add -f bridge/out.txt "$SNAP"
git commit -m "bridge: test pont codespace ($(date -u +%FT%TZ))" || true
for i in 1 2 3 4 5; do
  if git push origin HEAD:arena/01a061e4-vm-arena; then
    echo "PUBLICATION OK"
    break
  fi
  git pull --rebase origin arena/01a061e4-vm-arena || true
  sleep 5
done
exit 0
